```kotlin
// domain/repository/BnccRepository.kt (Interface)
interface BnccRepository {
    fun getEtapas(): Flow<Resource<List<Etapa>>>
    fun getSeriesByEtapa(etapaId: String): Flow<Resource<List<Serie>>>
    fun getEixosByEtapa(etapaId: String): Flow<Resource<List<Eixo>>>
    fun getEixosByEtapaAndSerie(etapaId: String, serieId: String): Flow<Resource<List<Eixo>>>
    fun getHabilidadesByEixo(etapaId: String, eixoId: String): Flow<Resource<List<Habilidade>>>
    fun getHabilidadesBySerieAndEixo(etapaId: String, serieId: String, eixoId: String): Flow<Resource<List<Habilidade>>>
    fun getHabilidadesByCompetencia(etapaId: String, competenciaId: String): Flow<Resource<List<Habilidade>>>
    fun getCompetencias(): Flow<Resource<List<Competencia>>>
    suspend fun syncAll(): Result<Unit>
}

// data/repository/BnccRepositoryImpl.kt (Implementação)
class BnccRepositoryImpl @Inject constructor(
    private val localDataSource: LocalDataSource,
    private val remoteDataSource: FirestoreDataSource,
    private val networkUtil: NetworkUtil,
    private val syncManager: SyncManager,
    private val habilidadeMapper: HabilidadeMapper,
    private val etapaMapper: EtapaMapper
) : BnccRepository {
    
    override fun getHabilidadesBySerieAndEixo(
        etapaId: String,
        serieId: String,
        eixoId: String
    ): Flow<Resource<List<Habilidade>>> = flow {
        
        emit(Resource.Loading())
        
        try {
            // 1. Buscar dados locais (SEMPRE primeiro - Offline First)
            localDataSource.getHabilidadesBySerieAndEixo(etapaId, serieId, eixoId)
                .map { entities ->
                    entities.map { habilidadeMapper.entityToDomain(it) }
                }
                .collect { localData ->
                    emit(Resource.Success(localData))
                    
                    // 2. Sincronizar em background se online
                    if (networkUtil.isConnected() && syncManager.shouldSync()) {
                        syncHabilidadesInBackground(etapaId)
                    }
                }
                
        } catch (e: Exception) {
            emit(Resource.Error(e.localizedMessage ?: "Erro desconhecido"))
        }
    }
    
    private suspend fun syncHabilidadesInBackground(etapaId: String) {
        withContext(Dispatchers.IO) {
            try {
                val remoteHabilidades = remoteDataSource.getHabilidadesByEtapa(etapaId)
                val entities = remoteHabilidades.map { habilidadeMapper.dtoToEntity(it) }
                localDataSource.insertHabilidades(entities)
                syncManager.updateLastSyncTime()
            } catch (e: Exception) {
                // Log error but don't crash
                Log.e("BnccRepository", "Sync error", e)
            }
        }
    }
    
    override suspend fun syncAll(): Result<Unit> = withContext(Dispatchers.IO) {
        try {
            if (!networkUtil.isConnected()) {
                return@withContext Result.failure(Exception("Sem conexão com a internet"))
            }
            
            // Sincronizar todas as entidades
            syncEtapas()
            syncSeries()
            syncEixos()
            syncCompetencias()
            syncHabilidades()
            
            syncManager.updateLastSyncTime()
            Result.success(Unit)
            
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
    
    private suspend fun syncHabilidades() {
        val remote = remoteDataSource.getAllHabilidades()
        val entities = remote.map { habilidadeMapper.dtoToEntity(it) }
        localDataSource.insertHabilidades(entities)
    }
    
    // Métodos similares para outras entidades...
}
```

```kotlin
// presentation/ui/habilidades/HabilidadesViewModel.kt
@HiltViewModel
class HabilidadesViewModel @Inject constructor(
    private val getHabilidadesUseCase: GetHabilidadesUseCase,
    private val syncDataUseCase: SyncDataUseCase,
    savedStateHandle: SavedStateHandle
) : ViewModel() {
    
    // Navegação parameters
    private val etapaId: String = savedStateHandle["etapaId"] ?: ""
    private val serieId: String? = savedStateHandle["serieId"]
    private val eixoId: String = savedStateHandle["eixoId"] ?: ""
    private val competenciaId: String? = savedStateHandle["competenciaId"]
    
    // UI State
    private val _uiState = MutableStateFlow<HabilidadesUiState>(HabilidadesUiState.Loading)
    val uiState: StateFlow<HabilidadesUiState> = _uiState.asStateFlow()
    
    init {
        loadHabilidades()
    }
    
    private fun loadHabilidades() {
        viewModelScope.launch {
            when {
                // Educação Infantil
                etapaId == "EI" -> {
                    getHabilidadesUseCase.getObjetivosByEixo(etapaId, eixoId)
                }
                
                // Ensino Fundamental
                etapaId in listOf("EF1", "EF2") && serieId != null -> {
                    getHabilidadesUseCase.getHabilidadesBySerieAndEixo(etapaId, serieId, eixoId)
                }
                
                // Ensino Médio
                etapaId == "EM" && competenciaId != null -> {
                    getHabilidadesUseCase.getHabilidadesByCompetencia(etapaId, competenciaId)
                }
                
                else -> flowOf(Resource.Error("Parâmetros inválidos"))
                
            }.collect { resource ->
                _uiState.value = when (resource) {
                    is Resource.Loading -> HabilidadesUiState.Loading
                    is Resource.Success -> HabilidadesUiState.Success(resource.data)
                    is Resource.Error -> HabilidadesUiState.Error(resource.message)
                }
            }
        }
    }
    
    fun refresh() {
        viewModelScope.launch {
            _uiState.value = HabilidadesUiState.Syncing
            
            val result = syncDataUseCase.syncAll()
            
            result.fold(
                onSuccess = { loadHabilidades() },
                onFailure = { 
                    _uiState.value = HabilidadesUiState.Error(it.message ?: "Erro na sincronização")
                }
            )
        }
    }
}

// UI States
sealed class HabilidadesUiState {
    object Loading : HabilidadesUiState()
    object Syncing : HabilidadesUiState()
    data class Success(val habilidades: List<Habilidade>) : HabilidadesUiState()
    data class Error(val message: String) : HabilidadesUiState()
}
```

```kotlin
// presentation/ui/habilidades/HabilidadesActivity.kt
@AndroidEntryPoint
class HabilidadesActivity : AppCompatActivity() {
    
    private lateinit var binding: ActivityHabilidadesBinding
    private val viewModel: HabilidadesViewModel by viewModels()
    private lateinit var adapter: HabilidadesAdapter
    
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        binding = ActivityHabilidadesBinding.inflate(layoutInflater)
        setContentView(binding.root)
        
        setupRecyclerView()
        setupObservers()
        setupSwipeRefresh()
    }
    
    private fun setupRecyclerView() {
        adapter = HabilidadesAdapter { habilidade ->
            navigateToDetail(habilidade)
        }
        
        binding.recyclerView.apply {
            layoutManager = LinearLayoutManager(this@HabilidadesActivity)
            adapter = this@HabilidadesActivity.adapter
            addItemDecoration(DividerItemDecoration(context, DividerItemDecoration.VERTICAL))
        }
    }
    
    private fun setupObservers() {
        lifecycleScope.launch {
            repeatOnLifecycle(Lifecycle.State.STARTED) {
                viewModel.uiState.collect { state ->
                    when (state) {
                        is HabilidadesUiState.Loading -> showLoading()
                        is HabilidadesUiState.Syncing -> showSyncing()
                        is HabilidadesUiState.Success -> showSuccess(state.habilidades)
                        is HabilidadesUiState.Error -> showError(state.message)
                    }
                }
            }
        }
    }
    
    private fun setupSwipeRefresh() {
        binding.swipeRefresh.setOnRefreshListener {
            viewModel.refresh()
        }
    }
    
    private fun showLoading() {
        binding.apply {
            progressBar.isVisible = true
            recyclerView.isVisible = false
            errorView.isVisible = false
        }
    }
    
    private fun showSuccess(habilidades: List<Habilidade>) {
        binding.apply {
            progressBar.isVisible = false
            recyclerView.isVisible = true
            errorView.isVisible = false
            swipeRefresh.isRefreshing = false
        }
        adapter.submitList(habilidades)
    }
    
    private fun showError(message: String) {
        binding.apply {
            progressBar.isVisible = false
            recyclerView.isVisible = false
            errorView.isVisible = true
            swipeRefresh.isRefreshing = false
        }
        Toast.makeText(this, message, Toast.LENGTH_LONG).show()
    }
    
    private fun navigateToDetail(habilidade: Habilidade) {
        val intent = Intent(this, HabilidadeDetailActivity::class.java).apply {
            putExtra("habilidade_id", habilidade.id)
        }
        startActivity(intent)
    }
}
```

```kotlin
// data/remote/FirestoreDataSource.kt
class FirestoreDataSource @Inject constructor(
    private val firestore: FirebaseFirestore
) {
    
    suspend fun getAllHabilidades(): List<HabilidadeDto> = withContext(Dispatchers.IO) {
        try {
            val snapshot = firestore.collection("habilidades")
                .get()
                .await()
            
            snapshot.documents.mapNotNull { doc ->
                doc.toHabilidadeDto()
            }
        } catch (e: Exception) {
            throw FirestoreException("Erro ao buscar habilidades: ${e.message}")
        }
    }
    
    suspend fun getHabilidadesByEtapa(etapaId: String): List<HabilidadeDto> = 
        withContext(Dispatchers.IO) {
            try {
                val snapshot = firestore.collection("habilidades")
                    .whereEqualTo("etapa_id", etapaId)
                    .get()
                    .await()
                
                snapshot.documents.mapNotNull { it.toHabilidadeDto() }
            } catch (e: Exception) {
                throw FirestoreException("Erro ao buscar habilidades: ${e.message}")
            }
        }
    
    private fun DocumentSnapshot.toHabilidadeDto(): HabilidadeDto? {
        return try {
            HabilidadeDto(
                id = id,
                codigo = getString("codigo") ?: return null,
                etapaId = getString("etapa_id") ?: return null,
                serieId = getString("serie_id"),
                eixoId = getString("eixo_id"),
                competenciaId = getString("competencia_id"),
                objetoConhecimento1 = getString("objeto_conhecimento_1"),
                objetoConhecimento2 = getString("objeto_conhecimento_2"),
                habilidade = getString("habilidade") ?: return null,
                explicacao = getString("explicacao"),
                exemplos = getString("exemplos"),
                ordem = getLong("ordem")?.toInt() ?: 0,
                tipo = getString("tipo") ?: "habilidade"
            )
        } catch (e: Exception) {
            null
        }
    }
}
```

### **7. Sync Manager - Gerenciamento de Sincronização**

```kotlin
// util/SyncManager.kt
class SyncManager @Inject constructor(
    private val preferences: SharedPreferences
) {
    
    companion object {
        private const val KEY_LAST_SYNC = "last_sync_timestamp"
        private const val SYNC_INTERVAL_HOURS = 24L
    }
    
    fun shouldSync(): Boolean {
        val lastSync = getLastSyncTime()
        val now = System.currentTimeMillis()
        val hoursSinceLastSync = (now - lastSync) / (1000 * 60 * 60)
        
        return hoursSinceLastSync >= SYNC_INTERVAL_HOURS
    }
    
    fun getLastSyncTime(): Long {
        return preferences.getLong(KEY_LAST_SYNC, 0L)
    }
    
    fun updateLastSyncTime() {
        preferences.edit()
            .putLong(KEY_LAST_SYNC, System.currentTimeMillis())
            .apply()
    }
    
    fun formatLastSyncTime(): String {
        val lastSync = getLastSyncTime()
        if (lastSync == 0L) return "Nunca sincronizado"
        
        val sdf = SimpleDateFormat("dd/MM/yyyy HH:mm", Locale.getDefault())
        return sdf.format(Date(lastSync))
    }
}
```
